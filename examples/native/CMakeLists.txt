cmake_minimum_required(VERSION 3.25)
if(NOT RPLUSPLUS_MODULE_DIR)
  message(FATAL_ERROR "Set RPLUSPLUS_MODULE_DIR to the installed Rpp language module directory")
endif()
list(PREPEND CMAKE_MODULE_PATH "${RPLUSPLUS_MODULE_DIR}")
project(MyApp LANGUAGES CXX Rpp)
get_property(_languages GLOBAL PROPERTY ENABLED_LANGUAGES)
if(NOT "Rpp" IN_LIST _languages)
  message(FATAL_ERROR "Rpp language was not enabled")
endif()
add_library(rustpart STATIC value.rs)
add_executable(MyApp main.cpp extra.rs)
target_link_libraries(MyApp PRIVATE rustpart)
enable_testing()
add_test(NAME mixed-native COMMAND MyApp)
