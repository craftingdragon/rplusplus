cmake_minimum_required(VERSION 3.25)
list(PREPEND CMAKE_MODULE_PATH "${RPLUSPLUS_MODULE_DIR}")
project(CrateGraph LANGUAGES CXX Rpp)
add_library(base STATIC base.rs)
set_target_properties(base PROPERTIES RPP_CRATE_TYPE rlib RPP_CRATE_NAME base RPP_EDITION 2021)
add_library(side STATIC side.rs)
set_target_properties(side PROPERTIES RPP_CRATE_TYPE rlib RPP_CRATE_NAME side)
target_link_libraries(side PUBLIC base)
add_library(middle STATIC middle.rs)
set_target_properties(middle PROPERTIES RPP_CRATE_TYPE rlib RPP_CRATE_NAME middle)
target_link_libraries(middle PUBLIC base side)
add_library(bridge STATIC bridge.rs)
set_target_properties(bridge PROPERTIES RPP_CRATE_TYPE staticlib RPP_CRATE_NAME bridge)
target_link_libraries(bridge PRIVATE middle)
target_compile_options(bridge PRIVATE --error-format=json)
add_library(adapter STATIC adapter.cpp)
target_link_libraries(adapter PRIVATE bridge)
add_executable(CppMain main.cpp)
target_compile_definitions(CppMain PRIVATE "$<$<CONFIG:Debug>:EXPECTED=1>" "$<$<NOT:$<CONFIG:Debug>>:EXPECTED=0>")
target_link_libraries(CppMain PRIVATE adapter)
add_executable(RustMain app.rs worker.cpp)
set_target_properties(RustMain PROPERTIES RPP_CRATE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/app.rs")
target_link_libraries(RustMain PRIVATE middle)
target_compile_options(RustMain PRIVATE "$<$<COMPILE_LANGUAGE:Rpp>:--cfg=driver_flag>")
install(TARGETS bridge EXPORT BridgeTargets ARCHIVE DESTINATION lib)
install(EXPORT BridgeTargets DESTINATION lib/cmake/Bridge)
enable_testing()
add_test(NAME rust-main COMMAND RustMain)
add_test(NAME cpp-main COMMAND CppMain)
